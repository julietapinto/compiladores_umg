EZEQUIELAQUIINICIA() [
    // Declaración de variables
    int x = 5;
    float y;
    string nombre = "Ezequiel";

    // Asignaciones
    x = x + 10;
    y = 3.14;

    // Estructura
    STRUCT Persona [
        string nombre;
        int edad;
    ];

    // Función
    FUNCION int sumar(int a, int b) [
        RETORNA a + b;
    ]

    // Llamada de función
    int resultado = sumar(x, 20);

    // Condicional
    CHI_LO_HACE(CON x > 10) [
        IMPRIMIR("X es mayor a 10");
    ] 
    TONCES [
        IMPRIMIR("Este es el bloque entonces");
    ] 
    CHI_NO [
        IMPRIMIR("X no es mayor a 10");
    ]

    // Ciclo while
    MIENTRAS [CON x < 20] [
        x = x + 1;
    ]

    // Ciclo for
    PARA [i = 0; CON i < 5; i = i + 1] [
        IMPRIMIR(i);
    ]

    // Switch
    SWITCH (x) [
        CASE 10:
            IMPRIMIR("X vale 10");
        CASE 20:
            IMPRIMIR("X vale 20");
        DEFAULT:
            IMPRIMIR("X tiene otro valor");
    ]

    // Import
    IMPORT "libreria_ejemplo";

    // Retorna
    RETORNA resultado;
]
EZEQUIELAQUIFINALIZA()
